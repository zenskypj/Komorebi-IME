cmake_minimum_required(VERSION 3.22)
project(komorebi_zenz32 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(opt LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_OPENSSL)
 set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${ZENZ_LLAMA_SOURCE} llama EXCLUDE_FROM_ALL)
add_library(zenz32_core STATIC engine.cpp)
target_link_libraries(zenz32_core PUBLIC llama)
if(ANDROID)
 add_library(komorebi_zenz32 SHARED jni.cpp)
 target_link_libraries(komorebi_zenz32 PRIVATE zenz32_core)
 target_link_options(komorebi_zenz32 PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
 add_executable(zenz32_check check.cpp)
 target_link_libraries(zenz32_check PRIVATE zenz32_core)
endif()
